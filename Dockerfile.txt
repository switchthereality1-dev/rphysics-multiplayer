# Rphysics multiplayer — минимальный образ без зависимостей
FROM node:20-alpine
WORKDIR /app
COPY rphysics-server.js package.json ./
COPY rphystestwebgl.html ./
ENV PORT=8080
EXPOSE 8080
HEALTHCHECK --interval=30s --timeout=3s CMD wget -qO- http://127.0.0.1:8080/health || exit 1
CMD ["node", "rphysics-server.js"]
